%% ---------------------------------------------------------------------------
%% Visual breaks for the two recurring asides. Deliberately greyscale and
%% unframed, to stay close to the ETH layout.
%%
%% \runexample opens a running example with a rule and a run-in title. It is a
%% command rather than an environment on purpose: every running example in the
%% thesis contains figure floats, and a float cannot be placed inside a box.
%% \begin{remarkbox} wraps a remark in a light grey panel. Remarks are short and
%% never contain floats, so a panel is safe there.
%% ---------------------------------------------------------------------------
\newcommand{\runexample}{%
  \par\addvspace{\bigskipamount}%
  \noindent\rule{\linewidth}{0.4pt}%
  \par\nobreak\vspace{0.45em}%
  \noindent\textbf{Running example.}\enspace\ignorespaces}

%% ---------------------------------------------------------------------------
%% \given is for conditioning that appears alongside an = sign, where \mid would
%% put two competing relations in one formula. \mid is \mathrel{|}, so TeX pads it
%% with \thickmuskip = 5mu plus 5mu: stretchable glue that widens when a justified
%% line is stretched, which makes the same formula look different line to line.
%% \vert is an ordinary symbol and \, is a fixed 3mu space, so this never stretches.
%% Keep \mid inside \Pr(...), where the wide relation spacing is conventional.
%% ---------------------------------------------------------------------------
\newcommand{\given}{\,\vert\,}
